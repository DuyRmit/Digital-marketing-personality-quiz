/** * Superpower board API — Cloudflare Worker + D1 * * GET /posts -> { posts: [{ id, persona, text, created_at }] } (newest first, hidden ones excluded) * POST /posts -> body { persona, text, name? } -> { post } * DELETE /posts/:id -> hides a post. Needs header Authorization: Bearer * * Posts show the persona plus an optional nickname. No IPs or other personal data are stored. */ const PERSONAS = ['creator', 'inclusive', 'trust', 'explorer', 'strategist', 'changemaker']; const MAX_LEN = 140; const MAX_NAME = 20; const LIST_LIMIT = 100; // Starter blocklists. Extend these before the event. // EN is matched on accent-stripped lowercase words; VI is matched on lowercase words keeping accents // (stripping accents would block normal Vietnamese words). const BLOCKED_EN = [ 'fuck', 'fucking', 'shit', 'bitch', 'cunt', 'dick', 'asshole', 'slut', 'nigger', 'faggot', ]; const BLOCKED_VI = ['đụ', 'địt', 'lồn', 'cặc', 'đéo', 'đm', 'đcm', 'vcl']; function stripAccents(s) { return s .normalize('NFD') .replace(/[\u0300-\u036f]/g, '') .replace(/đ/g, 'd'); } function contentProblem(t) { if (/(https?:\/\/|www\.|\S+@\S+\.\S+|(\d[\s.\-]?){8,})/i.test(t)) { return 'Please leave out links, emails and phone numbers.'; } const lower = t.toLowerCase(); const viWords = lower.split(/[^\p{L}\p{N}]+/u); const enWords = stripAccents(lower).split(/[^a-z0-9]+/); if (viWords.some((w) => BLOCKED_VI.includes(w)) || enWords.some((w) => BLOCKED_EN.includes(w))) { return 'Please keep it friendly and try rewording.'; } return ''; } function textProblem(raw) { const t = (raw ?? '').toString().trim(); if (!t) return 'Write one action first.'; if (t.length > MAX_LEN) return `Keep it under ${MAX_LEN} characters.`; return contentProblem(t); } function cleanName(raw) { return (raw ?? '') .toString() .replace(/[\u0000-\u001f]/g, ' ') .replace(/\s+/g, ' ') .trim(); } function nameProblem(raw) { const n = cleanName(raw); if (!n) return ''; if (n.length > MAX_NAME) return `Keep the name under ${MAX_NAME} characters.`; return contentProblem(n); } // Remembers the list for 5 seconds so many phones polling at once don't each hit the database. let cache = { at: 0, body: null }; function corsHeaders(request, env) { const allowed = (env.ALLOWED_ORIGIN || '*').split(',').map((s) => s.trim()); const origin = request.headers.get('Origin') || ''; const allow = allowed.includes('*') ? '*' : allowed.includes(origin) ? origin : allowed[0]; return { 'Access-Control-Allow-Origin': allow, 'Access-Control-Allow-Methods': 'GET, POST, DELETE, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization', 'Access-Control-Max-Age': '86400', Vary: 'Origin', }; } function json(data, status, cors) { return new Response(JSON.stringify(data), { status, headers: { 'Content-Type': 'application/json', 'Cache-Control': 'no-store', ...cors }, }); } export default { async fetch(request, env) { const cors = corsHeaders(request, env); if (request.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors }); const { pathname } = new URL(request.url); try { if (pathname === '/posts' && request.method === 'GET') { if (cache.body && Date.now() - cache.at < 5000) return json(cache.body, 200, cors); const { results } = await env.DB.prepare( 'SELECT id, persona, name, text, created_at FROM posts WHERE hidden = 0 ORDER BY id DESC LIMIT ?', ) .bind(LIST_LIMIT) .all(); cache = { at: Date.now(), body: { posts: results } }; return json(cache.body, 200, cors); } if (pathname === '/posts' && request.method === 'POST') { const body = await request.json().catch(() => null); if (!body || !PERSONAS.includes(body.persona)) return json({ error: 'Something went wrong. Please try again.' }, 400, cors); const problem = textProblem(body.text) || nameProblem(body.name); if (problem) return json({ error: problem }, 400, cors); const text = body.text.trim(); const name = cleanName(body.name); const created_at = Date.now(); const res = await env.DB.prepare( 'INSERT INTO posts (persona, name, text, created_at) VALUES (?, ?, ?, ?)', ) .bind(body.persona, name, text, created_at) .run(); cache.at = 0; return json( { post: { id: res.meta.last_row_id, persona: body.persona, name, text, created_at } }, 201, cors, ); } const del = pathname.match(/^\/posts\/(\d+)$/); if (del && request.method === 'DELETE') { const auth = request.headers.get('Authorization') || ''; if (!env.ADMIN_TOKEN || auth !== `Bearer ${env.ADMIN_TOKEN}`) return json({ error: 'Unauthorized' }, 401, cors); await env.DB.prepare('UPDATE posts SET hidden = 1 WHERE id = ?').bind(Number(del[1])).run(); cache.at = 0; return json({ ok: true }, 200, cors); } return json({ error: 'Not found' }, 404, cors); } catch (e) { return json({ error: 'Server error. Please try again.' }, 500, cors); } }, };
